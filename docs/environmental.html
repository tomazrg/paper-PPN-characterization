   dplyr::select(incidence,sand,silt,clay,ec,ph,cropping_corrigido,tillage_corrigido,altitute,soil_corrigido)


set.seed(123)
lambdas <- 10^seq(2, -3, by = -.1)

  dplyr::select(incidence) %>%

X <- all_genus%>%

  as.matrix()


# Setting alpha = 1 implements lasso regression
